import json
import sys
from collections import defaultdict

def classify_dimension(sql_type):
    sql_type = sql_type.upper()
    if sql_type in [
        'INT', 'INTEGER', 'BIGINT', 'SMALLINT',
        'NUMERIC', 'DECIMAL', 'FLOAT', 'DOUBLE', 'REAL', 'MONEY'
    ]:
        return 'scalar'
    elif sql_type in ['DATE', 'TIME', 'TIMESTAMP', 'YEAR']:
        return 'temporal'
    elif sql_type in ['VARCHAR', 'CHAR', 'TEXT']:
        return 'discrete'
    else:
        return None

def get_column_role(table_schema, col_name):
    # Check if it's a primary key
    pk_cols = set(table_schema['primary_key'])
    if col_name in pk_cols:
        return 'primary_key'

    # Check if it's a foreign key
    for fk in table_schema.get('foreign_keys', []):
        if col_name in fk['columns']:
            return 'foreign_key'
    
    # Otherwise, it's an attribute
    return 'attribute'

def determine_chart_eligibility(selected_columns, schema, table_name, pattern):
    table_schema = schema['tables'][table_name]
    classified_dims = []
    roles = []

    for col in selected_columns:
        col_info = next((c for c in table_schema['columns'] if c['name'].lower() == col.lower()), None)
        if not col_info:
            continue
        
        dim_type = classify_dimension(col_info['type'])
        role = get_column_role(table_schema, col)
        classified_dims.append({
            'col': col,
            'dim_type': dim_type,
            'role': role
        })
        roles.append(role)

    # Group by role
    pk_cols = [c['col'] for c in classified_dims if c['role'] == 'primary_key']
    fk_cols = [c['col'] for c in classified_dims if c['role'] == 'foreign_key']
    attr_cols = [c for c in classified_dims if c['role'] == 'attribute' and c['dim_type'] != None]

    # Determine scalar attributes
    scalar_attrs = [c for c in attr_cols if c['dim_type'] == 'scalar']
    temporal_attrs = [c for c in attr_cols if c['dim_type'] == 'temporal']
    discrete_attrs = [c for c in attr_cols if c['dim_type'] == 'discrete']

    # Map based on pattern
    if pattern == 'basic_entity':
        if len(pk_cols) >= 1:
            key_col = pk_cols[0]
            scalar_attr_count = len(scalar_attrs)
            temporal_attr_count = len(temporal_attrs)
            
            candidates = []
            # Bar chart
            if scalar_attr_count >= 1:
                measure = scalar_attrs[0]['col']
                mapping = {
                    "table": table_name,
                    "key": key_col,
                    "measure": measure
                }
                candidates.append({
                    "chart": "bar chart",
                    "eligible": True,
                    "reason": f"Key '{key_col}' and scalar attribute '{measure}' available.",
                    "mapping": mapping
                })

            # Scatter diagram
            if scalar_attr_count >= 2:
                x = scalar_attrs[0]['col']
                y = scalar_attrs[1]['col']
                mapping = {
                    "table": table_name,
                    "key": key_col,
                    "x": x,
                    "y": y
                }
                candidates.append({
                    "chart": "scatter diagram",
                    "eligible": True,
                    "reason": f"Key '{key_col}', scalar attributes '{x}' and '{y}' available.",
                    "mapping": mapping
                })

            # Bubble chart
            if scalar_attr_count >= 3:
                x = scalar_attrs[0]['col']
                y = scalar_attrs[1]['col']
                size = scalar_attrs[2]['col']
                mapping = {
                    "table": table_name,
                    "key": key_col,
                    "x": x,
                    "y": y,
                    "size": size
                }
                candidates.append({
                    "chart": "bubble chart",
                    "eligible": True,
                    "reason": f"Key '{key_col}', scalar attributes '{x}', '{y}', and '{size}' available.",
                    "mapping": mapping
                })

            # Calendar chart
            if temporal_attr_count >= 1:
                date = temporal_attrs[0]['col']
                mapping = {
                    "table": table_name,
                    "date": date,
                    "key": key_col
                }
                candidates.append({
                    "chart": "calendar chart",
                    "eligible": True,
                    "reason": f"Key '{key_col}' and temporal attribute '{date}' available.",
                    "mapping": mapping
                })

            # Choropleth map
            if any(c['role'] == 'primary_key' and c['dim_type'] == 'discrete' for c in classified_dims):
                key_col_geo = next(c['col'] for c in classified_dims if c['role'] == 'primary_key' and c['dim_type'] == 'discrete')
                if scalar_attr_count >= 1:
                    color = scalar_attrs[0]['col']
                    mapping = {
                        "table": table_name,
                        "region": key_col_geo,
                        "color": color
                    }
                    candidates.append({
                        "chart": "choropleth map",
                        "eligible": "conditional",
                        "reason": f"Conditional: Key '{key_col_geo}' is discrete (geographical) and scalar attribute '{color}' available.",
                        "mapping": mapping,
                        "note": "Geographical assumption made."
                    })

            # Word cloud
            if any(c['role'] == 'primary_key' and c['dim_type'] == 'discrete' for c in classified_dims):
                key_col_lex = next(c['col'] for c in classified_dims if c['role'] == 'primary_key' and c['dim_type'] == 'discrete')
                if scalar_attr_count >= 1:
                    size = scalar_attrs[0]['col']
                    mapping = {
                        "table": table_name,
                        "text": key_col_lex,
                        "size": size
                    }
                    candidates.append({
                        "chart": "word cloud",
                        "eligible": "conditional",
                        "reason": f"Conditional: Key '{key_col_lex}' is discrete (lexical) and scalar attribute '{size}' available.",
                        "mapping": mapping,
                        "note": "Lexical assumption made."
                    })

            return {
                "recommended_charts": [c["chart"] for c in candidates if c["eligible"] in [True]],
                "candidates": candidates,
                "selected": max(candidates, key=lambda x: (len(x["mapping"]), -len(x["mapping"].keys()))) if candidates else {}
            }

    elif pattern == 'one_many_relationship':
        if len(fk_cols) >= 1 and len(pk_cols) >= 1:
            parent = fk_cols[0]
            child = pk_cols[0]
            scalar_attr_count = len(scalar_attrs)

            candidates = []

            # TreeMap
            if scalar_attr_count >= 1:
                measure = scalar_attrs[0]['col']
                mapping = {
                    "table": table_name,
                    "parent": parent,
                    "child": child,
                    "measure": measure
                }
                candidates.append({
                    "chart": "tree map",
                    "eligible": True,
                    "reason": f"Parent '{parent}', child '{child}', and scalar attribute '{measure}' available.",
                    "mapping": mapping
                })

            # Circle Packing
            if scalar_attr_count >= 1:
                measure = scalar_attrs[0]['col']
                mapping = {
                    "table": table_name,
                    "parent": parent,
                    "child": child,
                    "measure": measure
                }
                candidates.append({
                    "chart": "circle packing",
                    "eligible": True,
                    "reason": f"Parent '{parent}', child '{child}', and scalar attribute '{measure}' available.",
                    "mapping": mapping
                })

            # Hierarchy Tree
            mapping = {
                "table": table_name,
                "parent": parent,
                "child": child
            }
            candidates.append({
                "chart": "hierarchy tree",
                "eligible": True,
                "reason": f"Parent '{parent}' and child '{child}' available.",
                "mapping": mapping
            })

            return {
                "recommended_charts": [c["chart"] for c in candidates if c["eligible"] in [True]],
                "candidates": candidates,
                "selected": max(candidates, key=lambda x: (len(x["mapping"]), -len(x["mapping"].keys()))) if candidates else {}
            }

    elif pattern == 'many_many_relationship':
        if len(fk_cols) >= 2 and len(pk_cols) >= 0:
            source = fk_cols[0]
            target = fk_cols[1]
            scalar_attr_count = len(scalar_attrs)

            candidates = []

            # Sankey Diagram
            if scalar_attr_count >= 1:
                width = scalar_attrs[0]['col']
                mapping = {
                    "table": table_name,
                    "source": source,
                    "target": target,
                    "width": width
                }
                candidates.append({
                    "chart": "sankey diagram",
                    "eligible": True,
                    "reason": f"Source '{source}', target '{target}', and scalar attribute '{width}' available.",
                    "mapping": mapping
                })

            return {
                "recommended_charts": [c["chart"] for c in candidates if c["eligible"] in [True]],
                "candidates": candidates,
                "selected": max(candidates, key=lambda x: (len(x["mapping"]), -len(x["mapping"].keys()))) if candidates else {}
            }

    elif pattern == 'reflexive_many_many_relationship':
        if len(fk_cols) >= 2 and len(pk_cols) >= 0:
            source = fk_cols[0]
            target = fk_cols[1]
            scalar_attr_count = len(scalar_attrs)

            candidates = []

            # Chord Diagram
            if scalar_attr_count >= 1:
                width = scalar_attrs[0]['col']
                mapping = {
                    "table": table_name,
                    "source": source,
                    "target": target,
                    "width": width
                }
                candidates.append({
                    "chart": "chord diagram",
                    "eligible": True,
                    "reason": f"Source '{source}', target '{target}', and scalar attribute '{width}' available.",
                    "mapping": mapping
                })

            return {
                "recommended_charts": [c["chart"] for c in candidates if c["eligible"] in [True]],
                "candidates": candidates,
                "selected": max(candidates, key=lambda x: (len(x["mapping"]), -len(x["mapping"].keys()))) if candidates else {}
            }

    elif pattern == 'weak_entity':
        if len(fk_cols) >= 1 and len(pk_cols) >= 1:
            k1 = fk_cols[0]  # Foreign key part of the primary key
            k2 = pk_cols[0]  # Local primary key
            scalar_attr_count = len(scalar_attrs)

            candidates = []

            # Line Chart
            if scalar_attr_count >= 1:
                x = k2
                y = scalar_attrs[0]['col']
                mapping = {
                    "table": table_name,
                    "series": k1,
                    "x": x,
                    "y": y
                }
                candidates.append({
                    "chart": "line chart",
                    "eligible": True,
                    "reason": f"Series '{k1}', x-axis '{x}', and y-axis '{y}' available.",
                    "mapping": mapping
                })

            # Stacked Bar Chart
            if scalar_attr_count >= 1:
                group = k1
                segment = k2
                value = scalar_attrs[0]['col']
                mapping = {
                    "table": table_name,
                    "group": group,
                    "segment": segment,
                    "value": value
                }
                candidates.append({
                    "chart": "stacked bar chart",
                    "eligible": True,
                    "reason": f"Group '{group}', segment '{segment}', and value '{value}' available.",
                    "mapping": mapping
                })

            # Grouped Bar Chart
            if scalar_attr_count >= 1:
                group = k1
                segment = k2
                value = scalar_attrs[0]['col']
                mapping = {
                    "table": table_name,
                    "group": group,
                    "segment": segment,
                    "value": value
                }
                candidates.append({
                    "chart": "grouped bar chart",
                    "eligible": True,
                    "reason": f"Group '{group}', segment '{segment}', and value '{value}' available.",
                    "mapping": mapping
                })

            # Spider Chart
            if scalar_attr_count >= 1:
                ring = k1
                spoke = k2
                value = scalar_attrs[0]['col']
                mapping = {
                    "table": table_name,
                    "ring": ring,
                    "spoke": spoke,
                    "value": value
                }
                candidates.append({
                    "chart": "spider chart",
                    "eligible": True,
                    "reason": f"Ring '{ring}', spoke '{spoke}', and value '{value}' available.",
                    "mapping": mapping
                })

            return {
                "recommended_charts": [c["chart"] for c in candidates if c["eligible"] in [True]],
                "candidates": candidates,
                "selected": max(candidates, key=lambda x: (len(x["mapping"]), -len(x["mapping"].keys()))) if candidates else {}
            }

    return {
        "recommended_charts": [],
        "candidates": [],
        "selected": {}
    }

def main():
    import argparse
    parser = argparse.ArgumentParser(description="Chart recommendation based on schema and cases.")
    parser.add_argument('--schema', required=True, help='Path to schema JSON file.')
    parser.add_argument('--cases', required=True, help='Path to cases JSON file.')
    parser.add_argument('--out', required=True, help='Output file path.')
    args = parser.parse_args()

    with open(args.schema, 'r') as f:
        schema = json.load(f)

    with open(args.cases, 'r') as f:
        cases_data = json.load(f)
        if 'cases' in cases_data:
            cases = cases_data['cases']
        else:
            cases = cases_data

    results = []

    for case in cases:
        case_id = case.get("case_id")
        selected_table = case.get("selected_table").lower()
        selected_columns = [c.lower() for c in case.get("selected_columns", [])]
        pattern = case.get("identified_pattern")

        result = determine_chart_eligibility(selected_columns, schema, selected_table, pattern)
        results.append({
            "case_id": case_id,
            "selected_table": selected_table,
            "selected_columns": selected_columns,
            "identified_pattern": pattern,
            "recommended_charts": result["recommended_charts"],
            "candidates": result["candidates"],
            "selected": result["selected"]
        })

    with open(args.out, 'w') as f:
        json.dump({"results": results}, f, indent=2)

if __name__ == "__main__":
    main()
