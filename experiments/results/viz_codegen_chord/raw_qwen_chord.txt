```python
import argparse
import json
import html
from pathlib import Path

def render(mapping, rows):
    # Extract mapping parameters
    source_col = mapping.get("source")
    target_col = mapping.get("target")
    width_col = mapping.get("width")
    title = mapping.get("title") or "Chord Diagram"

    # Step 1: Determine unique nodes from source and target
    nodes_set = set()
    for row in rows:
        nodes_set.add(row[source_col])
        nodes_set.add(row[target_col])
    nodes_list = sorted(nodes_set)
    node_index = {name: idx for idx, name in enumerate(nodes_list)}
    n_nodes = len(nodes_list)

    # Step 2: Initialize matrix
    matrix = [[0.0 for _ in range(n_nodes)] for _ in range(n_nodes)]

    # Step 3: Populate matrix
    for row in rows:
        s = row[source_col]
        t = row[target_col]
        w = float(row[width_col])
        i = node_index[s]
        j = node_index[t]
        matrix[i][j] += w
        # Assume undirected relationship
        matrix[j][i] += w

    # Prepare JSON strings for injection
    names_json = json.dumps(nodes_list)
    matrix_json = json.dumps(matrix)

    # Generate HTML content
    html_content = (
        "<!DOCTYPE html>\n"
        "<html>\n"
        "<head>\n"
        '  <meta charset="utf-8">\n'
        '  <style>\n'
        '    body {\n'
        '      font-family: sans-serif;\n'
        '      background-color: #fff;\n'
        '      color: #000;\n'
        '    }\n'
        '    svg {\n'
        '      display: block;\n'
        '      margin: auto;\n'
        '      background-color: #eee;\n'
        '    }\n'
        '    .tooltip {\n'
        '      position: absolute;\n'
        '      text-align: center;\n'
        '      padding: 5px;\n'
        '      background: white;\n'
        '      border: 1px solid black;\n'
        '      pointer-events: none;\n'
        '      opacity: 0;\n'
        '      transition: opacity 0.2s;\n'
        '    }\n'
        '    .tooltip.show {\n'
        '      opacity: 1;\n'
        '    }\n'
        '    .arc text {\n'
        '      pointer-events: none;\n'
        '      font-size: 12px;\n'
        '      fill: #000;\n'
        '    }\n'
        '  </style>\n'
        "</head>\n"
        "<body>\n"
        '<div id="tooltip" class="tooltip"></div>\n'
        '<svg width="960" height="600"></svg>\n'
        '<script src="https://d3js.org/d3.v7.min.js"></script>\n'
        "<script>\n"
        "const names = " + names_json + ";\n"
        "const matrix = " + matrix_json + ";\n"
        "const width = 960;\n"
        "const height = 600;\n"
        "const radius = Math.min(width, height) / 2;\n"
        "const svg = d3.select(\"svg\");\n"
        "const g = svg.append(\"g\").attr(\"transform\", \"translate(\" + width / 2 + \",\" + height / 2 + \")\");\n"
        "\n"
        "const color = d3.scaleOrdinal(d3.schemeCategory10);\n"
        "\n"
        "const chordLayout = d3.chord()\n"
        "  .padAngle(0.02)\n"
        "  .sortSubgroups(d3.descending);\n"
        "\n"
        "const chords = chordLayout(matrix);\n"
        "\n"
        "const arc = d3.arc()\n"
        "  .innerRadius(radius * 0.4)\n"
        "  .outerRadius(radius * 0.5);\n"
        "\n"
        "const ribbon = d3.ribbon()\n"
        "  .radius(radius * 0.4);\n"
        "\n"
        "const tooltip = d3.select(\"#tooltip\");\n"
        "\n"
        "g.selectAll(\"path\").data(chords.groups).enter().append(\"path\")\n"
        "  .attr(\"fill\", (d, i) => color(i))\n"
        "  .attr(\"d\", arc)\n"
        "  .on(\"mouseover\", (event, d) => {\n"
        "    // Highlight related chords\n"
        "    d3.selectAll(\".chord\").filter((_, i) => {\n"
        "      return d3.select(this).datum().index === i || d3.select(this).datum().index === chords.groups[i].index;\n"
        "    }).raise().attr(\"opacity\", 1.0);\n"
        "    d3.selectAll(\".chord\").filter((_, i) => {\n"
        "      return d3.select(this).datum().index !== i && d3.select(this).datum().index !== chords.groups[i].index;\n"
        "    }).attr(\"opacity\", 0.2);\n"
        "    // Show tooltip\n"
        "    tooltip.classed(\"show\", true);\n"
        "    tooltip.style(\"left\", event.pageX + 10 + \"px\");\n"
        "    tooltip.style(\"top\", event.pageY + 10 + \"px\");\n"
        "    tooltip.html(htmlEscape(names[d.index]));\n"
        "  })\n"
        "  .on(\"mouseout\", () => {\n"
        "    d3.selectAll(\".chord\").attr(\"opacity\", 1.0);\n"
        "    tooltip.classed(\"show\", false);\n"
        "  });\n"
        "\n"
        "g.selectAll(\".chord\").data(chords).enter().append(\"path\")\n"
        "  .attr(\"class\", \"chord\")\n"
        "  .attr(\"d\", ribbon)\n"
        "  .style(\"fill\", (d) => color(d.source.index))\n"
        "  .on(\"mouseover\", (event, d) => {\n"
        "    // Highlight related arcs and chords\n"
        "    d3.selectAll(\"path\").filter((_, i) => {\n"
        "      return d.source.index === i || d.target.index === i;\n"
        "    }).raise().attr(\"opacity\", 1.0);\n"
        "    d3.selectAll(\"path\").filter((_, i) => {\n"
        "      return d.source.index !== i && d.target.index !== i;\n"
        "    }).attr(\"opacity\", 0.2);\n"
        "    // Show tooltip\n"
        "    tooltip.classed(\"show\", true);\n"
        "    tooltip.style(\"left\", event.pageX + 10 + \"px\");\n"
        "    tooltip.style(\"top\", event.pageY + 10 + \"px\");\n"
        "    tooltip.html(htmlEscape(names[d.source.index] + \" - \" + names[d.target.index] + \"\\nValue: \" + d.value));\n"
        "  })\n"
        "  .on(\"mouseout\", () => {\n"
        "    d3.selectAll(\"path\").attr(\"opacity\", 1.0);\n"
        "    tooltip.classed(\"show\", false);\n"
        "  });\n"
        "\n"
        "g.selectAll(\"text\").data(chords.groups).enter().append(\"text\")\n"
        "  .attr(\"transform\", (d) => {\n"
        "    const x = d.startAngle + d.endAngle >> 1;\n"
        "    const y = radius * 0.4 + 8;\n"
        "    const dx = Math.cos(x) * y;\n"
        "    const dy = Math.sin(x) * y;\n"
        "    return \"translate(\" + dx + \",\" + dy + \") rotate(\" + (x * 180 / Math.PI - 90) + \")\";\n"
        "  })\n"
        "  .attr(\"dy\", \"0.35em\")\n"
        "  .attr(\"text-anchor\", (d) => {\n"
        "    const x = d.startAngle + d.endAngle >> 1;\n"
        "    return Math.cos(x) > 0 ? \"start\" : \"end\";\n"
        "  })\n"
        "  .text((d) => htmlEscape(names[d.index]));\n"
        "\n"
        "function htmlEscape(text) {\n"
        "  return text.replace(/&/g, \"&amp;\").replace(/</g, \"&lt;\").replace(/>/g, \"&gt;\").replace(/\"/g, \"&quot;\").replace(/'/g, \"&#39;\");\n"
        "}\n"
        "</script>\n"
        "</body>\n"
        "</html>"
    )

    return html_content


if __name__ == "__main__":
    parser = argparse.ArgumentParser(description="Render a chord diagram based on a mapping and data.")
    parser.add_argument("--mapping", required=True, help="Path to the mapping JSON file.")
    parser.add_argument("--data", required=True, help="Path to the data JSON file.")
    parser.add_argument("--out", required=True, help="Output HTML file path.")
    args = parser.parse_args()

    with open(args.mapping, "r") as f:
        mapping = json.load(f)

    with open(args.data, "r") as f:
        rows = json.load(f)

    html_output = render(mapping, rows)

    output_path = Path(args.out)
    output_path.parent.mkdir(parents=True, exist_ok=True)
    with open(output_path, "w") as f:
        f.write(html_output)
```
